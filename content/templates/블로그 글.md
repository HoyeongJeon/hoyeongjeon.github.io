<%*
// 가드: blog 폴더 밖에서 실행되면 중단한다.
// tp.file.rename() 은 "실행되는 파일" 의 이름을 바꾸므로, 이 템플릿 파일 자신에게
// 실행되면 템플릿이 스스로를 덮어써서 사라진다. 실제로 한 번 그렇게 날아갔다.
if (!tp.file.folder(true).startsWith("blog")) {
  throw new Error("이 템플릿은 blog 폴더 안에서만 실행할 수 있습니다.");
}

const title = await tp.system.prompt("글 제목 (한글 가능)");
const slug = await tp.system.prompt("파일명 (영문 slug — URL 이 된다)");
if (slug) await tp.file.rename(slug);
-%>
---
title: "<% title %>"
date: "<% tp.date.now('YYYY-MM-DD') %>"
tags: []
draft: true
---
